Backend Controller Architecture (Java Spring Boot)

Этот фрагмент Java кода обрабатывает запросы к API sapar:

@RestController
@RequestMapping("/api/v1")
@CrossOrigin(origins = "*")
public class SaparTourismController {

    @Autowired
    private CityTourismService cityService;

    @GetMapping("/cities/{id}")
    public ResponseEntity<CityDto> getCity(
            @PathVariable Long id,
            @RequestHeader(value = "Accept-Language", defaultValue = "ru") String lang) {
        return ResponseEntity.ok(cityService.getLocalizedCity(id, lang));
    }

    @GetMapping("/places")
    public ResponseEntity<List<PlaceDto>> getFilteredPlaces(
            @RequestParam(required = false) String category,
            @RequestParam(required = false) String city) {
        return ResponseEntity.ok(cityService.getPlaces(category, city));
    }
}
